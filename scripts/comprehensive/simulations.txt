# Simulation configs for scripts/comprehensive/simulation.py
#
# Each [name] section is key = value (space-separated lists for multi-value keys).
# Grid keys (same idea as scripts/distance/simulation.py):
#   semi_axes, fovs, resolutions, distances,
#   num_earth_points, num_positions_per_point, num_spins_per_position, num_radials_per_spin
#
# Quaternion noise (applied before full pipeline; noisy quat passed to pipeline_runner):
#   quaternion_noise_sigma_rad = <float>   (0 = off; default in CLI matches distance script)
#
# Distance-stage regression:
#   regression = tls | ols | ridge | ransac
#   ridge_lambda, ransac_residual_threshold, ransac_max_iterations, ransac_min_samples
#
# Render (same batch path as limb.simulation.main: _calculate_conic_coeffs + process_simulation):
#   render_sigma = <float>       edge blur in pixels (>= 0; 0 = sharp edge; default CLI: 0.5)
#   render_batch_size = <int>    batch size per resolution group (default CLI: 500)
#
# Post-render sensor noise (limb.simulation.render.conic._apply_noise_pipeline), applied in order:
#   noise_gaussian = <mean> <sigma>     additive Gaussian in pixel units (default sigma in limb: 10)
#   noise_stars = <prob>               salt (white) fraction [0,1], pepper 0
#   noise_discretization = <levels>    quantize to LEVELS levels per channel (>= 1)
#   noise_motion_blur = <kernel>       horizontal motion blur, odd kernel >= 1
#   noise_dead_pixels = <salt> <pepper> salt-and-pepper probabilities (last in pipeline)
#   noise_config_json = <path>         JSON object; keys gaussian | stars | discretization |
#                                      motion_blur | dead_pixels → nested objects (see limb doc).
#                                      Any noise_* lines above override that key when both set.
#
# Edge detection (Zernike + Sobel stage in full pipeline):
#   gray_threshold = <0-255>  Sobel high threshold (default 10)
#   window_size = <int>      Zernike window (default 7)
#   transition_width = <float> (default 1.66)
#
# Outputs: --output CSV and --images-dir (default <output_stem>_images) with img_<index>.png per row.
#
# Quaternion: same --quaternion w x y z as distance simulation (noisy_* columns). pipeline_runner
#   uses it without extra conjugation for --pipeline full (distance mode conjugates inside the binary).
#
# Profiling / tooling (CLI or boolean in section):
#   time_pipeline = true   → per-row runtime_sec (wall clock, perf_counter)
#   valgrind = true        → run each row under valgrind (requires valgrind on PATH; scale timeout)
